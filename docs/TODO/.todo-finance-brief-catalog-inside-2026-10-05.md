## 改动 2: 新增 scripts/run-on-octosense.py 启动包装

- 内容（草稿）:
  ```python
  #!/usr/bin/env python3
  # run-on-octosense.py — launch OctoSense shell with finance-brief catalog.
  #
  # Per OctoSense/desktop/README.md: shell reads --apps <path> first, else
  # ~/.octosense/apps.json, else config/apps.json. We use --apps to keep
  # every file under C:\Code\OctoSenseorg\ (per finance-brief policy).
  import argparse, os, subprocess, sys, time
  from pathlib import Path
  from urllib.request import urlopen

  DEFAULT_PORT = 57450

  def parse_args():
      p = argparse.ArgumentParser(prog="run-on-octosense.py")
      p.add_argument("--port", type=int, default=DEFAULT_PORT)
      p.add_argument("--no-wait", action="store_true",
                     help="don't wait for /s before returning")
      return p.parse_args()

  def get(port, route):
      with urlopen(f"http://127.0.0.1:{port}/{route}", timeout=2) as r:
          return r.read().decode("utf-8", "replace")

  def main():
      args = parse_args()
      here = Path(__file__).resolve().parent
      finance_brief = here.parent
      workspace = finance_brief.parent
      catalog = finance_brief / "catalog" / "apps.json"
      if not catalog.is_file():
          print(f"missing catalog: {catalog}", file=sys.stderr)
          print("run `python scripts/install-as-makepad-app.py` first",
                file=sys.stderr)
          raise SystemExit(2)
      shell_bin = (workspace / "OctoSense" / "target" / "release"
                    / "octosense.exe")
      if not shell_bin.is_file():
          print(f"missing shell binary: {shell_bin}", file=sys.stderr)
          raise SystemExit(3)

      port = args.port
      env = os.environ.copy()
      env.pop("OCTOSENSE_HOME", None)
      env["MAKEPAD_HIDE_WINDOWS"] = "1"
      env["MAKEPAD_REMOTE"] = str(port)
      env["RUSTUP_TOOLCHAIN"] = "stable-x86_64-pc-windows-msvc"

      print(f"launching: {shell_bin}")
      print(f"  --apps {catalog}")
      print(f"  MAKEPAD_REMOTE={port}")
      proc = subprocess.Popen(
          [str(shell_bin), "--apps", str(catalog)],
          cwd=str(workspace / "OctoSense"),
          env=env,
          creationflags=subprocess.CREATE_NO_WINDOW,
      )
      if not args.no_wait:
          for _ in range(60):
              try:
                  get(port, "s")
                  break
              except Exception:
                  time.sleep(0.5)
          else:
              print("warning: /s timeout", file=sys.stderr)
      print(f"octosense pid={proc.pid} port={port}")
      print("press Ctrl-C to stop")
      try:
          proc.wait()
      except KeyboardInterrupt:
          proc.terminate()
          proc.wait()

  if __name__ == "__main__":
      main()
  ```
- 理由: shell 不读 OctoSenseorg 外的 catalog；用 `--apps` 是唯一不修 dep 的办法
