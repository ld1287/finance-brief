## 3. 改动 2：新增 clean-shell-pollution.py

- 文件: `finance-brief/scripts/clean-shell-pollution.py`（新）
- 设计原则:
  - stdlib only（argparse / json / shutil / pathlib / sys / os）
  - 跨平台（pathlib）
  - 默认行为：诊断 → 清理 3 处 → 验证（一次跑完）
  - 支持子命令细化：`--diagnose-only`（只诊断不清理）/ `--dry-run`（打印动作但不写）/ `--user-cache-only`（只清 ~/.octosense/apps/.system/）/ `--octosense-only`（只清 OctoSense 侧）
- 草稿（结构骨架）：
  ```python
  #!/usr/bin/env python3
  """clean-shell-pollution.py — clear finance-brief pollution from OctoSense shell.

  Three pollution sources may exist after running install-as-system-app.py
  (legacy Page-format script) without --uninstall:

  1. OctoSense/desktop/system-apps.json (catalog entry)
  2. OctoSense/apps/finance-brief/        (partial bundle copy)
  3. ~/.octosense/apps/.system/os.finance-brief/<hash>/  (shell-cached pack)

  This script diagnoses all three, cleans them, and verifies a clean state.
  Use --diagnose-only to inspect without modifying anything.
  """
  import argparse, json, shutil, sys
  from pathlib import Path

  WS = Path(__file__).resolve().parent.parent.parent
  OCTO = WS / "OctoSense"
  OCTO_SEL = OCTO / "desktop" / "system-apps.json"
  OCTO_BUNDLE = OCTO / "apps" / "finance-brief"

  USER_BASE = Path(os.environ.get("OCTOSENSE_HOME") or Path.home())
  USER_SYS = USER_BASE / ".octosense" / "apps" / ".system" / "os.finance-brief"

  ENTRY_ID = "finance-brief"   # system-apps.json id and shell .system dir name


  def _clean_octosense(dry: bool) -> dict:
      """Clean OctoSense-side pollution. Returns a report dict."""
      report = {"catalog": {}, "bundle": {}}
      if not OCTO_SEL.is_file():
          report["catalog"] = {"path": str(OCTO_SEL), "status": "missing (no-op)"}
      else:
          data = json.loads(OCTO_SEL.read_text(encoding="utf-8"))
          apps = data.get("apps", [])
          before = list(apps)
          apps[:] = [a for a in apps if a != ENTRY_ID]
          if before != apps:
              if not dry:
                  OCTO_SEL.write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
              report["catalog"] = {"path": str(OCTO_SEL), "before": before, "after": apps, "status": "cleaned" if not dry else "would-clean"}
          else:
              report["catalog"] = {"path": str(OCTO_SEL), "apps": apps, "status": "already-clean"}
      if OCTO_BUNDLE.exists():
          if not dry:
              shutil.rmtree(OCTO_BUNDLE)
          report["bundle"] = {"path": str(OCTO_BUNDLE), "status": "cleaned" if not dry else "would-clean"}
      else:
          report["bundle"] = {"path": str(OCTO_BUNDLE), "status": "missing (no-op)"}
      return report


  def _clean_user_cache(dry: bool) -> dict:
      """Clean user-side shell-cached pack. Returns a report dict."""
      if USER_SYS.exists():
          if not dry:
              shutil.rmtree(USER_SYS)
          return {"path": str(USER_SYS), "status": "cleaned" if not dry else "would-clean"}
      return {"path": str(USER_SYS), "status": "missing (no-op)"}


  def _diagnose() -> dict:
      """Inspect all three locations; no writes."""
      diag = {"octosense": {}, "user_cache": {}}
      if OCTO_SEL.is_file():
          try:
              data = json.loads(OCTO_SEL.read_text(encoding="utf-8"))
              apps = data.get("apps", [])
              polluted = ENTRY_ID in apps
              diag["octosense"]["catalog"] = {
                  "path": str(OCTO_SEL),
                  "apps": apps,
                  "polluted": polluted,
              }
          except Exception as e:
              diag["octosense"]["catalog"] = {"path": str(OCTO_SEL), "error": str(e)}
      else:
          diag["octosense"]["catalog"] = {"path": str(OCTO_SEL), "missing": True}
      diag["octosense"]["bundle"] = {
          "path": str(OCTO_BUNDLE),
          "exists": OCTO_BUNDLE.exists(),
      }
      diag["user_cache"]["path"] = str(USER_SYS)
      diag["user_cache"]["exists"] = USER_SYS.exists()
      if USER_SYS.exists():
          hash_dirs = [p for p in USER_SYS.iterdir() if p.is_dir()]
          diag["user_cache"]["hash_subdirs"] = [p.name for p in hash_dirs]
      diag["any_pollution"] = (
          diag["octosense"].get("catalog", {}).get("polluted", False)
          or diag["octosense"]["bundle"]["exists"]
          or diag["user_cache"]["exists"]
      )
      return diag


  def main():
      ap = argparse.ArgumentParser(
          description="Clean finance-brief pollution from OctoSense shell.",
      )
      ap.add_argument("--diagnose-only", action="store_true",
                      help="print diagnostic report and exit (no writes)")
      ap.add_argument("--dry-run", action="store_true",
                      help="print what would be done without writing")
      scope = ap.add_mutually_exclusive_group()
      scope.add_argument("--user-cache-only", action="store_true",
                         help="only clean ~/.octosense/apps/.system/os.finance-brief/")
      scope.add_argument("--octosense-only", action="store_true",
                         help="only clean OctoSense/desktop/system-apps.json + apps/finance-brief/")
      args = ap.parse_args()

      diag = _diagnose()
      print("=== diagnose ===")
      print(json.dumps(diag, ensure_ascii=False, indent=2))

      if args.diagnose_only:
          return 10 if diag["any_pollution"] else 0

      print()
      print("=== clean ===")
      cleaned = []
      if not args.user_cache_only:
          r = _clean_octosense(args.dry_run)
          print(json.dumps(r, ensure_ascii=False, indent=2))
          cleaned.extend([v.get("status", "?") for v in r.values()])
      if not args.octosense_only:
          r = _clean_user_cache(args.dry_run)
          print(json.dumps(r, ensure_ascii=False, indent=2))
          cleaned.append(r.get("status", "?"))

      print()
      print("=== verify ===")
      final = _diagnose()
      print(json.dumps(final, ensure_ascii=False, indent=2))
      return 0 if not final["any_pollution"] else 1


  if __name__ == "__main__":
      sys.exit(main())
  ```

## 4. 改动 3：新增 diagnose-shell-state.py

- 文件: `finance-brief/scripts/diagnose-shell-state.py`（新）
- 设计原则:
  - **只读**，绝不修改
  - 输出 JSON（便于 CI / shell 解析）
  - exit code: 0 (clean) / 10 (pollution) / 1 (error)
- 草稿（结构骨架）：
  ```python
  #!/usr/bin/env python3
  """diagnose-shell-state.py — read-only diagnostic for OctoSense shell state.

  Inspects:
  1. OctoSense/desktop/system-apps.json (catalog)
  2. OctoSense/apps/finance-brief/        (partial bundle)
  3. ~/.octosense/apps/.system/os.finance-brief/  (shell cache)
  4. ~/.octosense/apps.json               (user catalog; checks for finance_brief)

  Output: JSON to stdout. Exit codes:
    0 = clean
    10 = pollution found
    1 = error reading files
  """
  import json, os, sys
  from pathlib import Path

  WS = Path(__file__).resolve().parent.parent.parent
  OCTO = WS / "OctoSense"
  OCTO_SEL = OCTO / "desktop" / "system-apps.json"
  OCTO_BUNDLE = OCTO / "apps" / "finance-brief"
  OCTO_CFG = OCTO / "desktop" / "config" / "apps.json"

  USER_BASE = Path(os.environ.get("OCTOSENSE_HOME") or Path.home())
  USER_SYS = USER_BASE / ".octosense" / "apps" / ".system" / "os.finance-brief"
  USER_CATALOG = USER_BASE / ".octosense" / "apps.json"

  SYSTEM_ID = "finance-brief"     # for system-apps.json / .system dir
  USER_ID = "finance_brief"       # for user catalog (underscore)


  def _scan_json(path, id_key, id_value):
      if not path.is_file():
          return {"path": str(path), "exists": False}
      try:
          data = json.loads(path.read_text(encoding="utf-8"))
      except Exception as e:
          return {"path": str(path), "exists": True, "error": str(e)}
      if isinstance(data, list):
          found = any(isinstance(e, dict) and e.get(id_key) == id_value for e in data)
          return {"path": str(path), "exists": True, "found": found, "count": len(data)}
      return {"path": str(path), "exists": True, "schema": "non-list"}


  def main():
      report = {
          "octosense_system_apps": _scan_json(OCTO_SEL, "id", SYSTEM_ID),
          "octosense_apps_json":   _scan_json(OCTO_CFG, "id", SYSTEM_ID),
          "octosense_bundle":      {"path": str(OCTO_BUNDLE), "exists": OCTO_BUNDLE.exists()},
          "user_cache":            {"path": str(USER_SYS), "exists": USER_SYS.exists()},
          "user_catalog":          _scan_json(USER_CATALOG, "id", USER_ID),
      }
      report["any_pollution"] = (
          report["octosense_system_apps"].get("found", False)
          or report["octosense_bundle"]["exists"]
          or report["user_cache"]["exists"]
      )
      report["user_app_registered"] = report["user_catalog"].get("found", False)
      print(json.dumps(report, ensure_ascii=False, indent=2))
      return 10 if report["any_pollution"] else 0


  if __name__ == "__main__":
      try:
          sys.exit(main())
      except Exception as e:
          print(json.dumps({"error": str(e)}, ensure_ascii=False), file=sys.stderr)
          sys.exit(1)
  ```
