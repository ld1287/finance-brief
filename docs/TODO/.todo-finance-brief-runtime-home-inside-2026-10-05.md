```python
#!/usr/bin/env python3
# migrate-octosense-home.py — copy ~/.octosense/ to finance-brief/.octosense/.
#
# One-time migration: finance-brief policy keeps every file under OctoSenseorg\.
# Shell runtime data (approvals/, apps/, secrets/, wm/) used to live at
# ~/.octosense/ (i.e. C:\Users\<USER>\.octosense\); run-on-octosense.py now
# redirects it via OCTOSENSE_HOME=<finance-brief>/.octosense. This script
# copies existing state over so the user keeps approvals, weights, app
# metadata, theme settings.
#
# Usage:
#   python scripts/migrate-octosense-home.py            # copy
#   python scripts/migrate-octosense-home.py --dry-run  # echo only
import argparse, shutil, sys
from pathlib import Path


def parse_args():
    p = argparse.ArgumentParser(prog="migrate-octosense-home.py")
    p.add_argument("--dry-run", action="store_true")
    return p.parse_args()


def main():
    args = parse_args()
    src = Path.home() / ".octosense"
    here = Path(__file__).resolve().parent
    dst = here.parent / ".octosense"
    if not src.is_dir():
        print(f"source missing: {src}", file=sys.stderr)
        print("(nothing to migrate; run-on-octosense.py will create the new one)",
              file=sys.stderr)
        return 0
    if dst.exists() and any(dst.iterdir()):
        print(f"destination already populated: {dst}", file=sys.stderr)
        print("remove it first if you want a fresh copy", file=sys.stderr)
        return 1
    if args.dry_run:
        print(f"[dry-run] would copy {src} -> {dst}")
        for f in src.rglob("*"):
            rel = f.relative_to(src)
            print(f"  {f.stat().st_size:>10} {rel}")
        return 0
    dst.mkdir(parents=True, exist_ok=True)
    n_files = 0
    for f in src.rglob("*"):
        if f.is_file():
            rel = f.relative_to(src)
            target = dst / rel
            target.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(f, target)
            n_files += 1
    print(f"copied {n_files} files: {src} -> {dst}")
    print()
    print("next: verify shell works with the new home, then remove the old:")
    print(f"  rmdir /s /q \"{src}\"")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

- 理由: 一键迁移；--dry-run 让用户先看要拷什么
